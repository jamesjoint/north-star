# Per-User Scraping + New Job Sources Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Split the scraper pipeline so query-based job boards (jsearch, linkedin, indeed) run per-user with LLM-generated search queries, while browse-based scrapers run once globally; add Indeed, togethxr, and teamworkonline scrapers.

**Architecture:** Global scrapers (fixed category pages) run once and feed a shared job pool. User scrapers (search-API-based) run per active user using `scraper_queries` cached in `profiles`. Queries are generated by Claude from the user's `background` text at onboarding and regenerated when background changes.

**Tech Stack:** Node.js ESM, PostgreSQL (pg), Playwright + stealth, Axios, Anthropic SDK (claude-haiku-4-5-20251001), Vitest, RapidAPI (jsearch + indeed)

- [ ] **Step 3: Implement query-generator.js**

Create `server/query-generator.js`:

```js
import Anthropic from '@anthropic-ai/sdk'
import { extractJSON } from './utils.js'

/**
 * Given a candidate's background text, ask Claude to generate
 * job search queries for scraper use.
 *
 * Returns: { terms: string[], location: string }
 * - terms: 6-10 search strings (role + context, no location suffix)
 * - location: preferred job location extracted from background
 */
export async function generateScraperQueries(background) {
  const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY })

  const response = await client.messages.create({
    model: 'claude-haiku-4-5-20251001',
    max_tokens: 1024,
    system: `You are a job search query generator. Given a candidate's professional background, generate optimized search queries for job boards.

Rules:
- Generate 6–10 search terms. Each should be a job title or role + context phrase (e.g. "Head of Partnerships creator economy", "Chief of Staff media", "Creator Solutions Manager").
- Do NOT include location in the terms — location is returned separately.
- Extract the candidate's preferred location from the background text (city/region). If unclear, default to "New York, NY".
- Terms should reflect the candidate's target roles and sectors, seniority, and specific domain interests.
- Do not repeat the same role with only slight wording variation.

Respond ONLY with valid JSON. No preamble, no explanation, no backticks.

{
  "queries": {
    "terms": ["<search term 1>", "<search term 2>", ...],
    "location": "<city, state>"
  }
}`,
    messages: [{
      role: 'user',
      content: `Candidate background:\n\n${background}\n\nGenerate job search queries for this candidate.`,
    }],
  })

  const parsed = extractJSON(response.content[0].text)
  return parsed.queries
}
```

```js
import axios from 'axios'
import { detectRemote } from './utils.js'

export const type = 'user'

export async function scrape({ terms, location }) {
  if (!process.env.RAPIDAPI_KEY) {
    console.warn('JSearch scraper: RAPIDAPI_KEY not set, skipping')
    return []
  }

  const seen = new Set()
  const jobs = []

  for (const term of terms) {
    const query = `${term} ${location}`
    try {
      const { data } = await axios.get('https://jsearch.p.rapidapi.com/search', {
        params: { query, num_pages: '1' },
        headers: {
          'X-RapidAPI-Key': process.env.RAPIDAPI_KEY,
          'X-RapidAPI-Host': 'jsearch.p.rapidapi.com',
        },
      })
      console.log(`[jsearch] query="${query}" status=${data.status} jobs=${(data.data || []).length}`)
      for (const job of (data.data || [])) {
        const url = job.job_apply_link
        if (!url || seen.has(url)) continue
        seen.add(url)
        const jobLocation = [job.job_city, job.job_state].filter(Boolean).join(', ')
        jobs.push({
          title: job.job_title || '',
          company: job.employer_name || '',
          location: jobLocation,
          remote: job.job_is_remote ? 'remote' : detectRemote(jobLocation),
          url,
          source: 'jsearch',
          posted_at: job.job_posted_at_datetime_utc
            ? job.job_posted_at_datetime_utc.slice(0, 10)
            : null,
          description: job.job_description || '',
        })
      }
    } catch (err) {
      console.error(`JSearch scraper: query "${query}" failed:`, err.message)
    }
  }

  return jobs
}
```

```js
import { detectRemote } from './utils.js'

export const type = 'user'

const LINKEDIN_ACTION = process.env.LINKEDIN_ACTION || 'LINKEDIN_SEARCH_JOBS'

export async function scrape({ terms, location }) {
  let toolset
  try {
    const { OpenAIToolSet } = await import('composio-core')
    toolset = new OpenAIToolSet({ apiKey: process.env.COMPOSIO_API_KEY })
  } catch (err) {
    console.error('LinkedIn scraper: failed to load composio-core:', err.message)
    return []
  }

  console.log(`[linkedin] Using action: ${LINKEDIN_ACTION}`)
  const seen = new Set()
  const jobs = []

  for (const term of terms) {
    let result
    try {
      result = await toolset.executeAction({
        action: LINKEDIN_ACTION,
        params: {
          keywords: term,
          location,
          count: 25,
        },
        entityId: 'default',
      })
    } catch (err) {
      console.error(`LinkedIn scraper: executeAction failed for "${term}":`, err.message)
      continue
    }

    if (!result?.data?.elements) {
      console.warn(`LinkedIn scraper: unexpected response for "${term}"`, JSON.stringify(result).slice(0, 200))
      continue
    }

    for (const job of parse(result.data.elements)) {
      if (!job.url || seen.has(job.url)) continue
      seen.add(job.url)
      jobs.push(job)
    }
  }

  return jobs
}

export function parse(elements) {
  return elements.map(job => ({
    title: job.title || job.position || '',
    company: job.companyName || job.company?.name || '',
    location: job.formattedLocation || job.location || 'New York, NY',
    remote: detectRemote(job.workplaceType || job.formattedLocation || ''),
    url: job.applyUrl || (job.entityUrn
      ? `https://www.linkedin.com/jobs/view/${job.entityUrn?.split(':').pop()}/`
      : ''),
    source: 'linkedin',
    posted_at: job.listedAt
      ? new Date(job.listedAt).toISOString().slice(0, 10)
      : null,
    description: job.description?.text || '',
  })).filter(j => j.title && j.url)
}
```

```js
import axios from 'axios'
import { detectRemote } from './utils.js'

export const type = 'user'

export async function scrape({ terms, location }) {
  if (!process.env.INDEED_RAPIDAPI_KEY) {
    console.warn('Indeed scraper: INDEED_RAPIDAPI_KEY not set, skipping')
    return []
  }

  const host = process.env.INDEED_RAPIDAPI_HOST || 'indeed12.p.rapidapi.com'
  const seen = new Set()
  const jobs = []

  for (const term of terms) {
    try {
      const { data } = await axios.get(`https://${host}/search`, {
        params: { query: `${term}`, location, page_id: '1' },
        headers: {
          'X-RapidAPI-Key': process.env.INDEED_RAPIDAPI_KEY,
          'X-RapidAPI-Host': host,
        },
      })
      const hits = data.hits || data.jobs || data.data || []
      console.log(`[indeed] term="${term}" hits=${hits.length}`)
      for (const hit of hits) {
        const jobkey = hit.jobkey || hit.job_id || hit.id
        if (!jobkey || seen.has(jobkey)) continue
        seen.add(jobkey)
        const jobLocation = hit.location || location
        jobs.push({
          title: hit.title || '',
          company: hit.company || hit.employer || '',
          location: jobLocation,
          remote: detectRemote(jobLocation),
          url: hit.url || `https://www.indeed.com/viewjob?jk=${jobkey}`,
          source: 'indeed',
          posted_at: hit.date ? new Date(hit.date).toISOString().slice(0, 10) : null,
          description: hit.snippet || hit.description || '',
        })
      }
    } catch (err) {
      console.error(`Indeed scraper: term "${term}" failed:`, err.message)
    }
  }

  return jobs
}
```

```js
import * as cheerio from 'cheerio'
import { launch } from './browser.js'
import { detectRemote } from './utils.js'

export const type = 'global'

const URL = 'https://jobs.togethxr.com/jobs'

export function parse(html) {
  const $ = cheerio.load(html)
  const jobs = []
  $('.job-card').each((_, el) => {
    const titleEl = $(el).find('.job-title a').first()
    const href = titleEl.attr('href')
    const title = titleEl.text().trim()
    if (!title || !href || href === '#') return
    const company = $(el).find('.company-link').first().text().trim()
    const location = $(el).find('.job-location').first().text().trim()

      title,
      company,
      location,
      remote: detectRemote(location),
      url: href.startsWith('http') ? href : `https://jobs.togethxr.com${href}`,
      source: 'togethxr',
      posted_at: null,
      description: '',
    })
  })
  return jobs
}

export async function scrape() {
  const browser = await launch()
  const page = await browser.newPage()
  try {
    await page.goto(URL, { waitUntil: 'domcontentloaded', timeout: 30000 })
    await page.waitForSelector('.job-card', { timeout: 15000 }).catch(() => {})
    const html = await page.content()
    return parse(html)
  } finally {
    await page.close()
    await browser.close()
  }
}
```

```js
import { upsertJob, getJobById, makeId } from '../db.js'

async function runScrapers(pool, scrapers, queries = null) {
  const newIds = []
  for (const scraper of scrapers) {
    try {
      const jobs = queries ? await scraper.scrape(queries) : await scraper.scrape()
      for (const job of jobs) {
        const alreadyExists = !!(await getJobById(pool, makeId(job.source, job.url)))
        const id = await upsertJob(pool, job)
        if (id && !alreadyExists) newIds.push(id)
      }
    } catch (err) {
      console.error(`Scraper failed:`, err)
    }
  }
  return newIds
}

export async function runGlobalScrapers(pool, scrapers) {
  return runScrapers(pool, scrapers)
}

export async function runUserScrapers(pool, scrapers, queries) {
  return runScrapers(pool, scrapers, queries)
}

export async function runAllScrapers(pool, scrapers) {
```

```js
export async function runPipeline(pool) {
  console.log('[pipeline] Starting scrape...')
  const adminUserId = await getAdminUserId(pool)

  // Phase 1: Global scrapers run once for all users
  const globalIds = await runGlobalScrapers(pool, GLOBAL_SCRAPERS)

  const allUserIds = await getActiveUsersWithProfiles(pool)
  const orderedUserIds = adminUserId
    ? [adminUserId, ...allUserIds.filter(id => id !== adminUserId)]
    : allUserIds

  if (orderedUserIds.length === 0) {
    console.log('[pipeline] No users with profiles to score for.')
    if (adminUserId) {
      sendDigestNotification(pool, globalIds, adminUserId).catch(err =>
        console.error('[notify] Failed:', err.message)
      )
    }
    return { scraped: globalIds.length, scored: 0 }
  }

  let totalScored = 0
  for (const userId of orderedUserIds) {
    const profile = await loadProfile(pool, userId).catch(() => null)
    if (!profile) continue

    // Phase 2: User scrapers run per-user with their cached queries
    let userIds = []
    if (profile.scraper_queries) {
      userIds = await runUserScrapers(pool, USER_SCRAPERS, profile.scraper_queries)
    } else {
      console.warn(`[pipeline] No scraper_queries for user ${userId} — skipping user scrapers`)
    }

    const newIds = [...new Set([...globalIds, ...userIds])]
    const backlogIds = (await getUnscoredJobs(pool, userId)).filter(id => !newIds.includes(id))
    const toScore = [...newIds, ...backlogIds]
    if (toScore.length === 0) continue

    const state = getUserState(userId)
    state.running = true
    state.lastCount = null

    console.log(`[pipeline] Scoring ${toScore.length} jobs for user ${userId} (${newIds.length} new, ${backlogIds.length} backlog)...`)
    let scored = 0
    for (let i = 0; i < toScore.length; i++) {
      const id = toScore[i]
      try {
        const job = await getJobById(pool, id)
        if (!job) continue
        const result = await scoreJob(job, profile)
        await updateScore(pool, id, userId, result.aggregate, { ...result.dimensions, summary: result.summary })
        scored++
        if (i < toScore.length - 1) await sleep(400)
      } catch (err) {
        console.error(`[pipeline] FAILED id=${id} user=${userId}:`, err.message)
      }
    }
    state.running = false
    state.lastCount = scored
    state.lastTs = Date.now()
    totalScored += scored
    console.log(`[pipeline] User ${userId}: ${scored}/${toScore.length} scored.`)

    sendDigestNotification(pool, newIds, userId).catch(err =>
      console.error(`[notify] Failed for user ${userId}:`, err.message)
    )
  }

  console.log(`[pipeline] Done. ${globalIds.length} global scraped, ${totalScored} total scores written.`)
  return { scraped: globalIds.length, scored: totalScored }
}

export function getRescoreStatus(userId) {
  if (userId) return { ...getUserState(userId) }
  const running = Object.values(rescoreByUser).some(s => s.running)
  const latest = Object.values(rescoreByUser).reduce((best, s) => {
    if (!s.lastTs) return best
    return !best || s.lastTs > best.lastTs ? s : best
  }, null)
  return { running, lastCount: latest?.lastCount ?? null, lastTs: latest?.lastTs ?? null }
}

export async function rescoreAllJobs(pool, userId) {

  if (state.running) {
    console.log(`[rescoreAllJobs] Already running for user ${userId}, skipping.`)
    return
  }
  state.running = true
  state.lastCount = null
  state.lastTs = null
  const { rows } = await pool.query('SELECT id FROM jobs WHERE deleted = false')
  const ids = rows.map(r => r.id)
  console.log(`[rescoreAllJobs] Re-scoring ${ids.length} jobs for user ${userId}...`)
  let scored = 0
  for (let i = 0; i < ids.length; i++) {
    try {
      await scoreOneJob(pool, ids[i], userId)
      scored++
    } catch (err) {
      console.error(`[rescoreAllJobs] Failed id=${ids[i]}:`, err.message)
    }
    if (i < ids.length - 1) await sleep(300)
  }
  state.running = false
  state.lastCount = scored
  state.lastTs = Date.now()
  console.log(`[rescoreAllJobs] Done. ${scored}/${ids.length} rescored for user ${userId}.`)
}

async function sendDigestNotification(pool, newIds, userId) {
  if (newIds.length === 0 || !userId) return
  try {
    const { rows: userRows } = await pool.query(`SELECT email FROM "user" WHERE id = $1`, [userId])
    const email = userRows[0]?.email
    if (!email) return
    const placeholders = newIds.map((_, i) => `$${i + 1}`).join(',')
    const userParam = `$${newIds.length + 1}`
    const { rows } = await pool.query(
      `SELECT j.title, j.company, j.location, js.score_aggregate
       FROM jobs j
       LEFT JOIN job_scores js ON js.job_id = j.id AND js.user_id = ${userParam}
       WHERE j.id IN (${placeholders})
       ORDER BY js.score_aggregate DESC NULLS LAST
       LIMIT 5`,
      [...newIds, userId]
```
